The highlighter tags SQL types in three different ways and doesn't know Flink's own types, such as STRING, so every type name in upper case gets the same class. A block with a `function` attribute, such as the examples on a built-in function page, also marks each call of that function. */}} {{- $html := transform.Highlight (print .Inner "\n") .Type .Options -}} {{- with .Attributes.function -}} {{- if findRE `^\w+$` . -}} {{- $html = replaceRE (printf `(?i)(%s)` .) `${1}` $html -}} {{- $html = replaceRE (printf `(?i)(%s)(\()` .) `${1}${2}` $html -}} {{- end -}} {{- end -}} {{- $types := `ARRAY|BIGINT|BINARY|BITMAP|BOOLEAN|BYTES|CHAR|DATE|DEC|DECIMAL|DOUBLE|FLOAT|INT|INTEGER|INTERVAL|MAP|MULTISET|NUMERIC|RAW|REAL|ROW|SMALLINT|STRING|TIME|TIMESTAMP|TIMESTAMP_LTZ|TINYINT|VARBINARY|VARCHAR|VARIANT` -}} {{- $html = replaceRE (printf `(%s)` $types) `${1}` $html -}} {{- /* TIME in TIMESTAMP WITH LOCAL TIME ZONE is part of the type's keywords, not a type. */ -}} {{- $html = replaceRE `(LOCAL\s+)TIME` `${1}TIME` $html -}} {{- $html | safeHTML -}}
